cmake_minimum_required(VERSION 3.16)
project(follow_me_ros LANGUAGES CXX)

if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)

find_package(ament_cmake REQUIRED)
find_package(follow_me_core REQUIRED)
find_package(follow_me_msgs REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(ros_gz_interfaces REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc)

set(ROS_DEPS rclcpp std_msgs geometry_msgs nav_msgs sensor_msgs visualization_msgs follow_me_msgs ros_gz_interfaces)

foreach(node leader_node detector_node tracker_node follow_node monitor_node radar_node fusion_node intercept_node hud_node)
  add_executable(${node} src/${node}.cpp)
  target_include_directories(${node} PRIVATE include)
  target_link_libraries(${node} follow_me_core::follow_me_core)
  ament_target_dependencies(${node} ${ROS_DEPS})
  install(TARGETS ${node} DESTINATION lib/${PROJECT_NAME})
endforeach()
foreach(node detector_node hud_node)
  ament_target_dependencies(${node} cv_bridge)
  target_link_libraries(${node} ${OpenCV_LIBS})
endforeach()

install(DIRECTORY config DESTINATION share/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_ros_utils test/test_ros_utils.cpp)
  target_include_directories(test_ros_utils PRIVATE include)
  target_link_libraries(test_ros_utils follow_me_core::follow_me_core)
  ament_target_dependencies(test_ros_utils rclcpp nav_msgs sensor_msgs visualization_msgs)
endif()

ament_package()
